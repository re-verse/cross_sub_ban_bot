
// ---- CSBB bot metrics (textfile written by /opt/csbb-ops/csbb-metrics.py) ----
prometheus.exporter.unix "csbb_textfile" {
  set_collectors    = ["textfile"]
  textfile {
    directory = "/var/lib/csbb-metrics"
  }
}

prometheus.scrape "csbb" {
  targets    = prometheus.exporter.unix.csbb_textfile.targets
  forward_to = [prometheus.remote_write.local.receiver]
  scrape_interval = "60s"
  job_name   = "csbb"
}
